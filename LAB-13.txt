 {$match:{CITY:"Baroda"}},
 {$count:"BarodaCount"}

 {$match:{CITY:"Rajkot"}},
 {$group:{_id:null, avg:{ $avg:"$FEES"}}}

 {$group:{_id:{Dept:"$DEPARTMENT",Gender:"$GENDER"}, count:{ $sum:1}}}

])

 {$sort:{FEES:-1}},
 {$limit:5}
])

// 15. Avg fees >12000 by city

 {$group:{_id:"$CITY", avg:{ $avg:"$FEES"}}},
 {$match:{avg:{ $gt:12000}}}
])

// 16. CE or Mechanical total fees
db.Student.aggregate([
 {$match:{DEPARTMENT:{ $in:["CE","Mechanical"]}}},
 {$group:{_id:null, total:{ $sum:"$FEES"}}}
])

// 17. City+Dept+Gender max/min/avg
db.Student.aggregate([
 {$group:{
  _id:{City:"$CITY",Dept:"$DEPARTMENT",Gender:"$GENDER"},
  max:{ $max:"$FEES"},
  min:{ $min:"$FEES"},
  avg:{ $avg:"$FEES"}
 }}
])

// 18. Rajkot → Avg fees per dept
db.Student.aggregate([
 {$match:{CITY:"Rajkot"}},
 {$group:{_id:"$DEPARTMENT", avg:{ $avg:"$FEES"}}}
])

// 19. Group by sem → total & avg
db.Student.aggregate([
 {$group:{_id:"$SEM", total:{ $sum:"$FEES"}, avg:{ $avg:"$FEES"}}},
 {$sort:{total:-1}}
])

// 20. Top 3 cities by total fees
db.Student.aggregate([
 {$group:{_id:"$CITY", total:{ $sum:"$FEES"}}},
 {$sort:{total:-1}},
 {$limit:3}
])

db.Stock.aggregate([{ $match:{pe:{ $gt:20}}}])

// 8. Profit where sales >250000
db.Stock.aggregate([
 {$match:{sales:{ $gt:250000}}},
 {$group:{_id:null,total:{ $sum:"$profit"}}}
])

// 9. Project company & profit
db.Stock.aggregate([{ $project:{company:1,profit:1,_id:0}}])

// 10. EPS > avg EPS
db.Stock.aggregate([
 {$group:{_id:null,avgEPS:{ $avg:"$eps"}}}
])

// ==================== PART C ====================

// 1. Profit > 30000 count
db.Stock.aggregate([
 {$match:{profit:{ $gt:30000}}},
 {$count:"count"}
])

// 2. Profit by sector sorted
db.Stock.aggregate([
 {$group:{_id:"$sector",total:{ $sum:"$profit"}}},
 {$sort:{total:-1}}
])

// 3. Top 3 sales
db.Stock.aggregate([
 {$sort:{sales:-1}},
 {$limit:3}
])

// 4. Avg PE by sector
db.Stock.aggregate([
 {$group:{_id:"$sector",avg:{ $avg:"$pe"}}}
])

// 5. Sum sales & profit per company
db.Stock.aggregate([
 {$project:{company:1,total:{ $add:["$sales","$profit"]}}}
])

// 6. Sales <400000 sorted desc
db.Stock.aggregate([
 {$match:{sales:{ $lt:400000}}},
 {$sort:{sales:-1}}
])

// 7. Count per sector
db.Stock.aggregate([{ $group:{_id:"$sector",count:{ $sum:1}}}])

// 8. Avg ROE sales >200000
db.Stock.aggregate([
 {$match:{sales:{ $gt:200000}}},
 {$group:{_id:null,avg:{ $avg:"$roe"}}}
])

// 9. Max profit per sector
db.Stock.aggregate([
 {$group:{_id:"$sector",max:{ $max:"$profit"}}}
])

// 10. Total sales & count per sector
db.Stock.aggregate([
 {$group:{_id:"$sector",total:{ $sum:"$sales"},count:{ $sum:1}}}
])

// 11. Project where profit >20000
db.Stock.aggregate([
 {$match:{profit:{ $gt:20000}}},
 {$project:{company:1,profit:1,_id:0}}
])

// 12. Lowest ROE sorted asc
db.Stock.aggregate([{ $sort:{roe:1}}])
